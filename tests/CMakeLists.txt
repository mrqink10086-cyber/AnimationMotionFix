cmake_minimum_required(VERSION 3.21)
project(AnimationMotionFixCoreTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ENABLE_ASAN "Enable AddressSanitizer" OFF)
add_executable(amf_core_tests core_tests.cpp)
target_include_directories(amf_core_tests PRIVATE ../include)
if(MSVC)
	target_compile_options(amf_core_tests PRIVATE /W4 /permissive- /utf-8)
	if(ENABLE_ASAN)
		target_compile_options(amf_core_tests PRIVATE /fsanitize=address /Zi)
		target_link_options(amf_core_tests PRIVATE /INCREMENTAL:NO /DEBUG)
	endif()
endif()
enable_testing()
add_test(NAME amf_core_tests COMMAND amf_core_tests)
